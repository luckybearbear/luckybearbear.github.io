@Override  
public void afterBindData(EventObject e) {  
    List<CellStyle> cellStyles = new ArrayList<>();  
    // 获取单据列表控件  
    BillList billList = getView().getControl(BILLLISTID);  
  
    ListSelectedRowCollection currentListAllRowCollection = billList.getCurrentListAllRowCollection();  
    DynamicObjectCollection collection = billList.getListModel().getProvider().getQueryResult().getCollection();  
    for (int i = 0; i < currentListAllRowCollection.size(); i++) {  
        ListSelectedRow listSelectedRow = currentListAllRowCollection.get(i);  
        int rowKey = listSelectedRow.getRowKey();  
        String warnType = collection.get(i).getString(PxtcSpareInventoryConst.PXTC_INVENTORY_WARN);  
        String color;  
        switch (warnType) {  
            case "1":  
                color = "green";  
                break;  
            case "2":  
                color = "#f1c40f";  
                break;  
            case "3":  
                color = "red";  
                break;  
            default:  
                color = "black";  
                break;  
        }  
        CellStyle cellStyle = new CellStyle();  
        // 设置字段  
        cellStyle.setFieldKey(PxtcSpareInventoryConst.PXTC_INVENTORY_WARN);  
        // 行号  
        cellStyle.setRow(rowKey);  
        // 字体颜色  
        cellStyle.setForeColor("DarkBlue");  
        // 单元格背景  
        cellStyle.setForeColor(color);  
        cellStyles.add(cellStyle);  
    }  
  
    billList.setCellStyle(cellStyles);  
}
