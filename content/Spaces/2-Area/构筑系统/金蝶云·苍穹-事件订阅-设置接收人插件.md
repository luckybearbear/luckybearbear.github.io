### 代码示例

```java
public class AuditSendUsersEventServicePlugin implements IEventServicePlugin {  
  
    @Override  
    public Map<String, List<Long>> calculateUserIds(KDBizEvent evt) {  
        //类型转换  
        EntityEvent entityEvent = (EntityEvent) evt;  
        String businessKey = entityEvent.getBusinesskeys().get(0);  
        String entityNumber = entityEvent.getEntityNumber();  
        DynamicObject data = BusinessDataServiceHelper.loadSingle(businessKey, entityNumber);  
        String id = data.getString(Consts.ID);  
  
        Map<String, List<Long>> result = new HashMap<>();  
        // IPQC  
        DynamicObject ipqcUser = data.getDynamicObject(PxtcMoldUpConst.PXTC_IPQC_USER);  
        if (ipqcUser != null) {  
            long userId = ipqcUser.getLong(Consts.ID);  
            result.computeIfAbsent(id, key -> new ArrayList<>()).add(userId);  
        } 
        return result;  
    }  
}
```
